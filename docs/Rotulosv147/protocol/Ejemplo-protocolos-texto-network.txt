-Crear 1 ventana de 64x8 en x=0 y=0 (En una pantalla de 64x16) :
0xff,0xff,0xff,0xff,          //ID Code (No cambia)
0x15,0x00,0x00,0x00, //Longitud paquete hasta el checksum
0x68,                         //Tipo de paquete (No cambia)
0x32,                         //Tipo de tarjeta (No cambia)
0x01,                         //ID tarjeta (No cambia)
0x7b,                         //Tipo protocolo (No cambia)
0x01,                         //0x01 recive confirmacion de envio, 0x00 no recive
0x0a,0x00,0x00,0x00, //Longitud paquete Comando sin checksum
0x01,0x01,0x00,0x00,0x00,0x00,0x00,0x40,0x00,0x08,

//0x01 Comando crear ventana
//0x01 Numero de ventanas a crear
//0x00,0x00 Coordenada X
//0x00,0x00 Coordenada Y
//0x00,0x40 Ancho ventana
//0x00,0x08 Alto ventana

0x6b,0x01                       //0x6b,0x01 Checksum, suma de todos los bytes desde paquete de datos
                       hasta tipo de paquete. (16B = 0x6b,0x01)
